# Vue 常用交互动画效果

这里收集开发中常用的交互动画效果，直接拿过去就能用。

## 1. 防抖 Debounce

连续触发，停止 N 毫秒后才执行，适合搜索输入、窗口 resize。

```typescript
export function debounce<T extends (...args: any[]) => any>(
  fn: T,
  delay: number = 300
) {
  let timer: number | null = null
  return function(...args: Parameters<T>) {
    if (timer) clearTimeout(timer)
    timer = window.setTimeout(() => {
      fn(...args)
    }, delay)
  }
}
```

使用：
```typescript
const handleSearch = debounce((keyword: string) => {
  console.log('搜索', keyword)
  // 发请求
}, 300)

input.addEventListener('input', e => {
  handleSearch(e.target.value)
})
```

## 2. 节流 Throttle

规定时间内只执行一次，适合 scroll、mousemove。

```typescript
export function throttle<T extends (...args: any[]) => any>(
  fn: T,
  delay: number = 100
) {
  let last = 0
  return function(...args: Parameters<T>) {
    const now = Date.now()
    if (now - last >= delay) {
      fn(...args)
      last = now
    }
  }
}
```

使用：
```typescript
const onScroll = throttle(() => {
  console.log('滚动位置', window.scrollY)
}, 100)

window.addEventListener('scroll', onScroll)
```

## 3. 背景跟随鼠标倾斜（视差效果）

这个效果常用于首页英雄区，很炫酷，用户移动鼠标背景轻微倾斜。

```vue
<template>
  <div
    ref="containerRef"
    class="hero"
    @mousemove="handleMouseMove"
  >
    <div class="bg" :style="bgStyle"></div>
    <div class="content">
      <h1>欢迎来到我的网站</h1>
    </div>
  </div>
</template>

<script setup>
import { ref } from 'vue'

const containerRef = ref<HTMLElement | null>(null)
const rotateX = ref(0)
const rotateY = ref(0)

const handleMouseMove = (e: MouseEvent) => {
  if (!containerRef.value) return

  const rect = containerRef.value.getBoundingClientRect()

  // 鼠标在容器中的位置 -0.5 ~ 0.5
  const x = (e.clientX - rect.left) / rect.width - 0.5
  const y = (e.clientY - rect.top) / rect.height - 0.5

  // 最大倾斜角度，这里 5 度，可以调
  rotateY.value = x * 5
  rotateX.value = -y * 5
}

const bgStyle = computed(() => ({
  transform: `rotateX(${rotateX.value}deg) rotateY(${rotateY.value}deg)`
}))
</script>

<style scoped>
.hero {
  position: relative;
  width: 100%;
  height: 100vh;
  perspective: 1000px;
  overflow: hidden;
}

.bg {
  position: absolute;
  top: -5%;
  left: -5%;
  width: 110%;
  height: 110%;
  background: url('your-bg.jpg') center/cover;
  transition: transform 0.1s ease-out;
  will-change: transform;
  z-index: 0;
}

.content {
  position: relative;
  z-index: 1;
  color: white;
  text-align: center;
  padding-top: 200px;
}
</style>
```

效果：鼠标往哪动，背景往哪倾斜，轻微动，很有立体感。

## 4. 滚动吸顶

导航滚动到顶部吸住，常见于博客文档网站。

```vue
<template>
  <div
    ref="navRef"
    :class="['navbar', { 'is-sticky': isSticky }]"
  >
    <!-- 导航内容 -->
  </div>
</template>

<script setup>
import { ref, onMounted, onUnmounted } from 'vue'
import { throttle } from '../utils'

const navRef = ref<HTMLElement | null>(null)
const isSticky = ref(false)

const handleScroll = throttle(() => {
  if (!navRef.value) return
  const { top } = navRef.value.getBoundingClientRect()
  // top <= 0 说明已经滚到顶部了，吸住
  isSticky.value = top <= 0
}, 100)

onMounted(() => {
  window.addEventListener('scroll', handleScroll)
})

onUnmounted(() => {
  window.removeEventListener('scroll', handleScroll)
})
</script>

<style scoped>
.navbar.is-sticky {
  position: fixed;
  top: 0;
  left: 0;
  width: 100%;
  background: rgba(255, 255, 255, 0.95);
  box-shadow: 0 2px 8px rgba(0, 0, 0, 0.1);
  z-index: 999;
  animation: slideDown 0.3s ease;
}

@keyframes slideDown {
  from {
    transform: translateY(-100%);
  }
  to {
    transform: translateY(0);
  }
}
</style>
```

## 5. 回到顶部按钮

```vue
<template>
  <button
    v-show="showBackTop"
    @click="backToTop"
    class="back-top"
  >
    ↑
  </button>
</template>

<script setup>
import { ref, onMounted, onUnmounted } from 'vue'
import { throttle } from '../utils'

const showBackTop = ref(false)

const handleScroll = throttle(() => {
  showBackTop.value = window.scrollY > 300
}, 100)

const backToTop = () => {
  window.scrollTo({
    top: 0,
    behavior: 'smooth' // 平滑滚动
  })
}

onMounted(() => {
  window.addEventListener('scroll', handleScroll)
})

onUnmounted(() => {
  window.removeEventListener('scroll', handleScroll)
})
</script>

<style scoped>
.back-top {
  position: fixed;
  bottom: 30px;
  right: 30px;
  width: 40px;
  height: 40px;
  border-radius: 50%;
  background: #007bff;
  color: white;
  border: none;
  cursor: pointer;
  font-size: 20px;
  box-shadow: 0 2px 8px rgba(0, 0, 0, 0.15);
  z-index: 999;
  transition: all 0.3s;
}

.back-top:hover {
  background: #0056b3;
  transform: translateY(-2px);
}
</style>
```

## 6. 图片懒加载

原生已经支持，直接用：

```vue
<img
  :src="src"
  alt="图片"
  loading="lazy"
>
```

如果需要自定义懒加载（先占位，加载完淡入）：

```vue
<template>
  <div class="lazy-image">
    <img
      v-show="!loaded"
      class="placeholder"
      src="data:image/gif;base64,R0lGODlhAQABAIAAAAAAAP///yH5BAEAAAAALAAAAAABAAEAAAIBRAA7"
      :style="{ backgroundColor: placeholderColor }"
    />
    <img
      ref="imgRef"
      :data-src="src"
      :class="{ loaded }"
      class="real-img"
      alt
    />
  </div>
</template>

<script setup>
import { ref, onMounted } from 'vue'

interface Props {
  src: string
  placeholderColor?: string
}

const props = withDefaults(defineProps<Props>(), {
  placeholderColor: '#f5f5f5'
})

const imgRef = ref<HTMLImageElement | null>(null)
const loaded = ref(false)

onMounted(() => {
  if ('IntersectionObserver' in window && imgRef.value) {
    const io = new IntersectionObserver((entries) => {
      entries.forEach(entry => {
        if (entry.isIntersecting && imgRef.value) {
          imgRef.value.src = imgRef.value.dataset.src || ''
          imgRef.value.onload = () => {
            loaded.value = true
          }
          io.unobserve(entry.target)
        }
      })
    })
    io.observe(imgRef.value)
  }
})
</script>

<style scoped>
.lazy-image {
  position: relative;
  overflow: hidden;
}
.placeholder {
  width: 100%;
  height: 100%;
  display: block;
}
.real-img {
  width: 100%;
  transition: opacity 0.3s;
  opacity: 0;
}
.real-img.loaded {
  opacity: 1;
}
</style>
```

## 7. 下拉框点击外面关闭

```vue
<template>
  <div ref="dropdownRef" class="dropdown">
    <button @click="toggle">菜单</button>
    <div v-show="isOpen" class="dropdown-menu">
      <div class="item">选项1</div>
      <div class="item">选项2</div>
    </div>
  </div>
</template>

<script setup>
import { ref } from 'vue'
import { useClickOutside } from '../../hooks'

const dropdownRef = ref<HTMLElement | null>(null)
const isOpen = ref(false)

const toggle = () => {
  isOpen.value = !isOpen.value
}

useClickOutside(dropdownRef, () => {
  isOpen.value = false
})
</script>
```

useClickOutside hook 在「常用组合式函数」那篇有，直接用。

## 8. 水波纹点击效果

点击按钮扩散波纹动画，Material Design 常见效果：

```vue
<template>
  <button
    class="ripple-btn"
    @click="handleClick"
  >
    <span
      v-for="ripple in ripples"
      :key="ripple.id"
      class="ripple"
      :style="ripple.style"
    ></span>
    <slot></slot>
  </button>
</template>

<script setup>
import { ref } from 'vue'

interface Ripple {
  id: number
  x: number
  y: number
  size: number
}

const ripples = ref<Ripple[]>([])

const handleClick = (e: MouseEvent) => {
  const btn = e.currentTarget as HTMLElement
  const rect = btn.getBoundingClientRect()
  const size = Math.max(rect.width, rect.height)
  const x = e.clientX - rect.left - size / 2
  const y = e.clientY - rect.top - size / 2

  const ripple = {
    id: Date.now() + Math.random(),
    x,
    y,
    size
  }

  ripples.value.push(ripple)

  // 动画完删掉
  setTimeout(() => {
    ripples.value.shift()
  }, 600)
}
</script>

<style scoped>
.ripple-btn {
  position: relative;
  overflow: hidden;
  /* 按钮其他样式 */
}

.ripple {
  position: absolute;
  border-radius: 50%;
  background: rgba(255, 255, 255, 0.5);
  transform: scale(0);
  animation: ripple 0.6s ease-out;
  pointer-events: none;
  width: var(--size);
  height: var(--size);
  left: var(--x);
  top: var(--y);
}

@keyframes ripple {
  to {
    transform: scale(4);
    opacity: 0;
  }
}
</style>
```

使用 CSS variable 动态设置大小位置，完美。

## 9. 数字滚动动画

从 0 滚动到目标数字，类似数据看板计数动画：

```typescript
export function animateNumber(
  target: number,
  duration: number = 1000,
  callback: (value: number) => void
) {
  const start = 0
  const startTime = performance.now()

  function step(timestamp: number) {
    const progress = Math.min((timestamp - startTime) / duration, 1)
    // 缓动函数 easeOutCubic
    const ease = 1 - Math.pow(1 - progress, 3)
    const current = start + (target - start) * ease
    callback(current)
    if (progress < 1) {
      requestAnimationFrame(step)
    }
  }

  requestAnimationFrame(step)
}
```

vue 组件：

```vue
<template>
  <div class="count-up">{{ current }}</div>
</template>

<script setup>
import { ref, onMounted } from 'vue'
import { animateNumber } from './animateNumber'

interface Props {
  target: number
  duration?: number
}

const props = defineProps<Props>()
const current = ref(0)

onMounted(() => {
  animateNumber(props.target, props.duration || 1000, (val) => {
    current.value = Math.round(val)
  })
})
</script>
```

## 10. 平滑滚动到指定元素

```typescript
export function scrollToId(id: string, offset = 0) {
  const el = document.getElementById(id)
  if (!el) return
  const top = el.getBoundingClientRect().top + window.scrollY + offset
  window.scrollTo({
    top,
    behavior: 'smooth'
  })
}
```

## 总结

常用工具函数整理：
- `debounce` - 防抖 → 搜索输入
- `throttle` - 节流 → 滚动鼠标移动
- `useClickOutside` - 点击外部关闭 → 下拉弹框
- `useStorage` - 持久化存储 → 用户设置
- `animateNumber` - 数字滚动 → 数据看板
- 背景跟随鼠标倾斜 → 首页炫酷效果
- 滚动吸顶、回到顶部 → 导航常用
- 图片懒加载 → 性能优化
- 水波纹点击 → 交互反馈

这些都是项目中经常用到的，复制过去改改就能用。
