# Vue3 常用组合式函数（Hooks）

组合式函数就是把可复用的逻辑抽出来，useXxx 命名，在组件里直接用。

## 1. useMouse - 获取鼠标位置

```typescript
// hooks/useMouse.ts
import { ref, onMounted, onUnmounted } from 'vue'

export function useMouse() {
  const x = ref(0)
  const y = ref(0)

  function handleMove(e: MouseEvent) {
    x.value = e.pageX
    y.value = e.pageY
  }

  onMounted(() => {
    window.addEventListener('mousemove', handleMove)
  })

  onUnmounted(() => {
    window.removeEventListener('mousemove', handleMove)
  })

  return { x, y }
}
```

使用：
```vue
<script setup>
import { useMouse } from './hooks/useMouse'
const { x, y } = useMouse()
</script>

<template>
  <p>鼠标位置: {{ x }}, {{ y }}</p>
</template>
```

## 2. useInterval - 定时器

```typescript
// hooks/useInterval.ts
import { onUnmounted } from 'vue'

export function useInterval(
  callback: () => void,
  delay: number
) {
  const timer = setInterval(callback, delay)

  onUnmounted(() => {
    clearInterval(timer)
  })
}
```

使用：
```typescript
import { ref } from 'vue'
import { useInterval } from './hooks/useInterval'

const count = ref(0)
useInterval(() => {
  count.value++
}, 1000)
```

## 4. useStorage - localStorage/sessionStorage 响应式持久化

```typescript
// hooks/useStorage.ts
import { ref, watch } from 'vue'

export function useStorage<T>(
  key: string,
  defaultValue: T,
  storage: Storage = localStorage
) {
  // 读取初始值
  const getStoredValue = () => {
    const raw = storage.getItem(key)
    if (raw) {
      try {
        return JSON.parse(raw) as T
      } catch (e) {
        return defaultValue
      }
    }
    return defaultValue
  }

  const value = ref<T>(getStoredValue())

  // 变化了存起来
  watch(value, (newVal) => {
    if (newVal === undefined) {
      storage.removeItem(key)
    } else {
      storage.setItem(key, JSON.stringify(newVal))
    }
  }, { deep: true })

  return value
}
```

使用：
```typescript
// 响应式，自动存在 localStorage
const username = useStorage('username', '')
const theme = useStorage<'light' | 'dark'>('theme', 'light')
```

改了自动存，刷新页面还是那个值，完美！

## 5. useToggle - 布尔值切换

```typescript
// hooks/useToggle.ts
import { ref } from 'vue'

export function useToggle(initialValue = false) {
  const value = ref(initialValue)

  function toggle() {
    value.value = !value.value
  }

  function setTrue() {
    value.value = true
  }

  function setFalse() {
    value.value = false
  }

  return { value, toggle, setTrue, setFalse }
}
```

使用：
```typescript
const { value: isOpen, toggle } = useToggle(false)

// 点击切换
// <button @click="toggle">切换</button>
```

太简洁了！

## 6. useWindowSize - 获取窗口大小

```typescript
// hooks/useWindowSize.ts
import { ref, onMounted, onUnmounted } from 'vue'

export function useWindowSize() {
  const width = ref(window.innerWidth)
  const height = ref(window.innerHeight)

  function handleResize() {
    width.value = window.innerWidth
    height.value = window.innerHeight
  }

  onMounted(() => {
    window.addEventListener('resize', handleResize)
  })

  onUnmounted(() => {
    window.removeEventListener('resize', handleResize)
  })

  return { width, height }
}
```

使用：
```typescript
const { width } = useWindowSize()
// 响应式，变了自动更新
const isMobile = computed(() => width.value < 768)
```

## 7. useClickOutside - 点击外部触发

常用在弹框、下拉菜单，点击外面关闭：

```typescript
// hooks/useClickOutside.ts
import type { Ref } from 'vue'
import { onMounted, onUnmounted } from 'vue'

export function useClickOutside(
  el: Ref<HTMLElement | null>,
  callback: () => void
) {
  function handleClick(e: MouseEvent) {
    if (!el.value) return
    if (!el.value.contains(e.target as Node)) {
      callback()
    }
  }

  onMounted(() => {
    document.addEventListener('click', handleClick)
  })

  onUnmounted(() => {
    document.removeEventListener('click', handleClick)
  })
}
```

使用：
```vue
<script setup>
import { ref } from 'vue'
import { useClickOutside } from './hooks/useClickOutside'

const dropdownRef = ref<HTMLElement | null>(null)
const isOpen = ref(false)

useClickOutside(dropdownRef, () => {
  isOpen.value = false
})
</script>

<template>
  <div ref="dropdownRef">
    下拉内容
  </div>
</template>
```

## 8. useDebounce - 防抖

```typescript
// hooks/useDebounce.ts
export function useDebounce<T extends (...args: any[]) => any>(
  fn: T,
  delay: number
) {
  let timer: number | null = null
  return function(...args: Parameters<T>) {
    if (timer) clearTimeout(timer)
    timer = window.setTimeout(() => {
      fn(...args)
    }, delay)
  }
}
```

使用：
```typescript
const handleSearch = useDebounce((keyword: string) => {
  // 发请求搜索
}, 300)

input.addEventListener('input', e => {
  handleSearch(e.target.value)
})
```

## 9. useThrottle - 节流

```typescript
// hooks/useThrottle.ts
export function useThrottle<T extends (...args: any[]) => any>(
  fn: T,
  delay: number
) {
  let last = 0
  return function(...args: Parameters<T>) {
    const now = Date.now()
    if (now - last >= delay) {
      fn(...args)
      last = now
    }
  }
}
```

使用：
```typescript
const onScroll = useThrottle(() => {
  console.log('scroll')
}, 100)

window.addEventListener('scroll', onScroll)
```

## 10. useRequest - 请求封装（带加载、错误）

```typescript
// hooks/useRequest.ts
import { ref } from 'vue'

type RequestFn<T> = () => Promise<T>

export function useRequest<T>(request: RequestFn<T>) {
  const data = ref<T | null>(null)
  const loading = ref(false)
  const error = ref<Error | null>(null)

  async function run() {
    loading.value = true
    error.value = null
    try {
      data.value = await request()
      return data.value
    } catch (err) {
      error.value = err as Error
      throw err
    } finally {
      loading.value = false
    }
  }

  return { data, loading, error, run }
}
```

使用：
```typescript
const { data, loading, error, run } = useRequest(() =>
  fetch('/api/user').then(res => res.json())
)

// 调用
run()
```

## 总结

自定义 hook 好处：
- 逻辑抽离，复用方便，多个组件用直接导入就行
- 每个 hook 只做一件事，干净整洁
- 和组件逻辑分离，好测试
- 社区已经有 `vueuse` 这个库，现成的各种 hooks，项目直接用：

```bash
npm install @vueuse/core
```

```typescript
import { useMouse, useWindowSize, useLocalStorage } from '@vueuse/core'
// 直接用，不用自己写了
```

项目开发，自己业务逻辑抽成 hook，通用的直接用 vueuse，完美。
